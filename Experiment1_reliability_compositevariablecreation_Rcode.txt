# Recode condition variables to factors
supp <- supp %>% 
  mutate(condition = as.factor(condition),
         entitytype = as.factor(entitytype),
         mentalfaculty = as.factor(mentalfaculty),
         gender = as.factor(gender),
         raceethnicity = as.factor(raceethnicity),
         region = as.factor(region),
         education_recode = as.factor(education_recode),
         diet = as.factor(diet),
         religion = as.factor(religion),
         religionRNR = as.factor(religionRNR),
         own = as.factor(own),
         work = as.factor(work),
         smart = as.factor(smart)
         )

# Condition assignment numbers
supp %>% 
  group_by(condition) %>% 
  summarise(no_rows = length(condition))

supp %>% 
  group_by(entitytype) %>% 
  summarise(no_rows = length(entitytype))

supp %>% 
  group_by(mentalfaculty) %>% 
  summarise(no_rows = length(mentalfaculty))

# Variable Prep for Regressions

# Relevel factor predictors for largest group
supp <- supp %>%
  mutate(
    raceethnicity = fct_relevel(raceethnicity, "White"),
    region = fct_relevel(region, "South"),
    religion = fct_relevel(religion, "Protestant"),
    diet = fct_relevel(diet, "Meat-eater"),
    mentalfaculty = fct_relevel(mentalfaculty, "none"),
    entitytype = fct_relevel(entitytype, "robotAI"))


# Mean center predictors without a meaningful zero
forcen <- supp[, c("Pcode", "age", "politics", "IntC", "income")]
forcen[c(2:5)] <- lapply(forcen[c(2:5)], function(x) {
  meanCenter(x)
})
forcen <- dplyr::rename(forcen, agecen = "age", 
                        polcen = "politics",
                        intcen = "IntC",
                        incomecen = "income")
supp <- full_join(supp, forcen, by = "Pcode")
